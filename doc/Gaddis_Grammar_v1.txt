GRAMÁTICA

-- PROGRAMA --

programa ::= { instruccion } ;

-- INSTRUCCIONES --

instruccion ::=
      constante_declaracion
    | variable_declaracion
    | asignacion
    | entrada
    | salida
    | retorno
    | if_stmt
    | while_stmt
    | for_stmt
    | switch_stmt
    | funcion
    | procedimiento
    | llamada_procedimiento ;

-- BLOQUES --

bloque ::= { instruccion } ;

-- VARIABLES Y TIPOS --

variable_declaracion ::=
    "Declarar"
    tipo
    identificador
    { "," identificador }
    [ dimensiones ]
    ;

dimensiones ::=
    "[" expresion "]"
    { "[" expresion "]" }
    ;

constante_declaracion ::=
    "Constante"
    tipo
    identificador
    "<-"
    expresion ;

tipo ::=
      "Entero"
    | "Real"
    | "Cadena"
    | "Logico"
    | "Caracter" ;

-- ASIGNACIÓN --

asignacion ::=
    L_valor
    "<-"
    expresion ;
    
L_valor ::=
    identificador
    { "[" expresion "]" } ;

-- ENTRADA / SALIDA --

entrada ::=
    "Leer"
    identificador
    { "," identificador } ;

salida ::=
    "Escribir"
    expresion
    { "," expresion } ;

-- EXPRESIONES --

expresion ::= logica ;

logica ::= or_expr ;

or_expr ::=
    and_expr
    { "O" and_expr } ;

and_expr ::=
    not_expr
    { "Y" not_expr } ;

not_expr ::=
      "No" not_expr
    | comparacion ;

comparacion ::=
    suma
    [
        (
            "="
          | "<>"
          | "<"
          | ">"
          | "<="
          | ">="
        )
        suma
    ] ;

suma ::=
    termino
    {
        ("+" | "-")
        termino
    } ;

termino ::=
    potencia
    {
        ("*" | "/" | "MOD")
        potencia
    } ;

potencia ::=
    acceso
    [
        "^"
        potencia
    ] ;

primario ::=
      numero
    | cadena
    | logico
    | llamada_funcion
    | identificador
    | "(" expresion ")" ;
    
acceso ::=
    identificador
    { "[" expresion "]" } ;

logico ::=
      "Verdadero"
    | "Falso" ;

-- CONDICIONALES --

if_stmt ::=
    "Si"
    expresion
    "Entonces"

    bloque

    { else_if }

    [ else ]

    "Fin"
    "Si" ;

else_if ::=
    "Sino"
    "Si"
    expresion
    "Entonces"

    bloque ;

else ::=
    "Sino"

    bloque ;

-- MIENTRAS --

while_stmt ::=
    "Mientras"
    expresion

    bloque

    "Fin"
    "Mientras" ;

-- PARA --

for_stmt ::=
    "Para"
    identificador
    "<-"
    expresion

    "Hasta"
    expresion

    [ "Paso" expresion ]

    bloque

    "Fin"
    "Para" ;

-- SELECCIONAR CASO --

switch_stmt ::=
    "Seleccionar"
    expresion

    caso
    { caso }

    [ defecto ]

    "Fin"
    "Seleccionar" ;

caso ::=
    "Caso"
    expresion
    ":"

    bloque ;

defecto ::=
    "Defecto"
    ":"

    bloque ;

-- PROCEDIMIENTO --

procedimiento ::=
    "Procedimiento"
    identificador
    "("
    [ parametros ]
    ")"

    bloque

    "Fin"
    "Procedimiento" ;

-- FUNCIÓN --

funcion ::= 
    "Funcion"
    tipo
    identificador
    "(" [ parametros ] ")"

    bloque

    "Fin"
    "Funcion" ;

-- PARÁMETROS --

parametros_lista ::=
    parametro
    { "," parametro } ;

parametro ::=
    tipo
    identificador
    dimensiones_opc ;
    
dimensiones_opc ::=
    dimensiones
  | ε ;

-- LLAMADAS --

llamada_procedimiento ::=
    "Llamar"
    identificador
    "("
    [ argumentos ]
    ")" ;

llamada_funcion ::=
    identificador
    "("
    [ argumentos ]
    ")" ;

argumentos ::=
    expresion
    { "," expresion } ;

-- RETORNO --

retorno ::=
    "Retornar"
    expresion ;

-- IDENTIFICADORES Y LITERALES --

identificador ::=
    letra
    {
        letra
      | digito
      | "_"
    } ;

numero ::=
    digito
    { digito } ;

cadena ::=
    '"'
    { caracter }
    '"' ;

Semántica

Perfecto. Con estas decisiones ya podemos definir una **especificación semántica formal** del lenguaje antes de implementar `SemanticAnalyzer`. Esto es importante porque el analizador semántico no debe "inventar" reglas; debe aplicar exactamente la filosofía del lenguaje.

Voy a consolidar tus respuestas como reglas oficiales.

---

# Especificación semántica del lenguaje

## 1. Sistema de tipos

### 1.1 Tipos base oficiales

El lenguaje será tipo **Gaddis/PSeInt completo**:

Tipos primitivos:

| Tipo       | Descripción       |
| ---------- | ----------------- |
| `Entero`   | Números enteros   |
| `Real`     | Números decimales |
| `Cadena`   | Texto             |
| `Caracter` | Un solo carácter  |
| `Logico`   | Valores booleanos |
| `Nulo`     | Ausencia de valor |

Ejemplos:

```txt
Declarar edad Como Entero

Declarar promedio Como Real

Declarar nombre Como Cadena

Declarar inicial Como Caracter

Declarar activo Como Logico
```

---

# 1.2 Declaración de variables

El tipo siempre es obligatorio.

Correcto:

```txt
Declarar edad Como Entero
```

Incorrecto:

```txt
Declarar edad
```

El AST podría existir, pero semánticamente:

```
Error:
La variable 'edad' requiere un tipo explícito.
```

---

# 1.3 Tipado estático fuerte

Una variable tiene un tipo fijo durante toda su vida.

Ejemplo:

```txt
Declarar x Como Entero

x <- 10
```

Correcto.

Pero:

```txt
x <- "hola"
```

Error:

```
Error semántico:
No se puede asignar Cadena a Entero.
```

---

# 2. BinaryExpression

Aquí hay decisiones importantes.

---

# 2.1 Operador `+`

Tu lenguaje tendrá una regla híbrida:

## Suma numérica

```txt
5 + 3
```

Resultado:

```
Entero
```

---

## Concatenación cuando participa Cadena

Permitido:

```txt
"Edad: " + edad
```

Ejemplo:

```txt
Declarar Entero edad

edad <- 20

Escribir "Edad: " + edad
```

Resultado:

```
Edad: 20
```

---

La regla será:

| Operandos                         | Resultado |
| --------------------------------- | --------- |
| Entero + Entero                   | Entero    |
| Entero + Real                     | Real      |
| Real + Real                       | Real      |
| Cadena + cualquier tipo permitido | Cadena    |
| Logico + Entero                   | ❌ Error   |
| Logico + Real                     | ❌ Error   |

---

# 2.2 Comparaciones entre tipos

Comparaciones estrictas.

Ejemplo:

```txt
"5" > 3
```

Error:

```
No se pueden comparar Cadena y Entero.
```

Permitido:

```txt
5 > 3
```

```txt
"x" = "y"
```

---

# 2.3 Operadores lógicos

Solo aceptan:

```txt
Logico
```

Ejemplo válido:

```txt
(x > 5) Y (y < 10)
```

Porque:

```
x > 5 → Logico
y < 10 → Logico
```

---

Incorrecto:

```txt
5 Y "hola"
```

Error:

```
El operador Y requiere operandos Logico.
```

---

# 3. Conversión de tipos

Tu lenguaje tendrá **coerción limitada controlada**.

No será como JavaScript.

---

Permitido:

```txt
"Edad: " + edad
```

Regla especial:

```
Cadena + cualquier tipo básico
=
Cadena
```

---

No permitido:

```txt
5 + Verdadero
```

Error:

```
No existe conversión automática de Logico a Entero.
```

---

# 4. Variables

## 4.1 Variables no declaradas

Error obligatorio.

Ejemplo:

```txt
x <- 10
```

Sin:

```txt
Declarar x Como Entero
```

Resultado:

```
Variable 'x' no declarada.
```

---

# 4.2 Redeclaración

No permitida.

Ejemplo:

```txt
Declarar x Como Entero

Declarar x Como Real
```

Error:

```
La variable x ya fue declarada.
```

---

# 5. Scope

## 5.1 Scope de bloques

El lenguaje tendrá scope léxico.

Ejemplo:

```txt
Mientras x < 10

    Declarar temporal Como Entero

Fin Mientras


Escribir temporal
```

Error:

```
La variable temporal no existe fuera del bloque.
```

---

# 5.2 Shadowing

No permitido.

Ejemplo:

```txt
Declarar x Como Entero


Mientras condicion

    Declarar x Como Real

Fin Mientras
```

Error:

```
La variable x ya existe en un scope superior.
```

---

# 6. Control de flujo

## 6.1 Condiciones booleanas obligatorias

No hay coerción.

Incorrecto:

```txt
Mientras 5
```

Error:

```
Mientras requiere una expresión Logica.
```

Correcto:

```txt
Mientras x < 10
```

---

# 6.2 Return

Funciones:

```txt
Funcion suma() Como Entero

    Retornar 10

Fin Funcion
```

Obligatorio.

---

Procedimientos:

```txt
Procedimiento mostrar()

    Retornar

Fin Procedimiento
```

Puede tener retorno vacío.

Pero:

```txt
Procedimiento mostrar()

    Retornar 10

Fin Procedimiento
```

Error:

```
Un procedimiento no puede retornar un valor.
```

---

# 7. Entrada / salida

## 7.1 Escribir

Permite mezcla de tipos.

Ejemplo:

```txt
Escribir "Edad:", edad
```

Internamente:

```
Cadena + conversión controlada
```

---

También:

```txt
Escribir "Edad: " + edad
```

válido.

---

# 7.2 Leer

Respeta el tipo declarado.

Ejemplo:

```txt
Declarar edad Como Entero

Leer edad
```

El runtime espera:

```
Número entero
```

Si recibe:

```
hola
```

Error de ejecución:

```
No se puede convertir 'hola' a Entero.
```

---

# 8. Funciones y procedimientos

## 8.1 Retorno obligatorio

Una función siempre declara tipo:

Correcto:

```txt
Funcion suma() Como Entero
```

Incorrecto:

```txt
Funcion suma()
```

---

# 8.2 Return obligatorio

La función debe garantizar retorno.

Ejemplo:

```txt
Funcion mayor(a Entero, b Entero) Como Entero

    Si a > b Entonces
        Retornar a
    Fin Si

Fin Funcion
```

Error:

```
La función puede finalizar sin retornar valor.
```

# 10. Constantes (nuevo detalle)

Las constantes serán **inmutables**.

Ejemplo:

```txt
Declarar Constante pi = 3.1416
```

Registro:

```js
{
    name:"pi",
    kind:"constant",
    type:"Real",
    mutable:false
}
```

---

Esto:

```txt
pi <- 10
```

produce:

```
Error:
No se puede modificar la constante 'pi'.
```
